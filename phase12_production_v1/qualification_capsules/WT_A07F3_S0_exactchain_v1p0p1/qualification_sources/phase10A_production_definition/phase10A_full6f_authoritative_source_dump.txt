    13	def configure_generated_polarization_blocks(generated_root):
    14	    """Apply WHIZARD-3.1.8-compatible polarization blocks after card rendering.
    15	
    16	    Current campaign naming convention:
    17	      *_unpol_* : omit polarization include entirely
    18	      *_LR100_* : e- left,  e+ right => @(-1), @(+1)
    19	      *_RL100_* : e- right, e+ left  => @(+1), @(-1)
    20	    """
    21	    root = Path(generated_root)
    22	    if not root.exists():
    23	        return {"unpol_scrubbed": 0, "lr_written": 0, "rl_written": 0}
    24	
    25	    counts = {"unpol_scrubbed": 0, "lr_written": 0, "rl_written": 0}
    26	
    27	    for card in sorted(root.glob("**/process.sin")):
    28	        card_text = card.read_text(encoding="utf-8")
    29	        card_lines = card_text.splitlines()
    30	        card_str = str(card)
    31	
    32	        if "_unpol_" in card_str:
    33	            new_lines = [line for line in card_lines if "polarization.inc" not in line]
    34	            if new_lines != card_lines:
    35	                card.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
    36	                counts["unpol_scrubbed"] += 1
    37	            continue
    38	
    39	        pol = card.parent / "common" / "polarization.inc"
    40	
    41	        if "_LR100_" in card_str:
    42	            pol.write_text(
    43	                "beams_pol_density = @(-1), @(+1)\n"
    44	                "beams_pol_fraction = 100%, 100%\n",
    45	                encoding="utf-8",
    46	            )
    47	            counts["lr_written"] += 1
    48	
    49	        elif "_RL100_" in card_str:
    50	            pol.write_text(
    51	                "beams_pol_density = @(+1), @(-1)\n"
    52	                "beams_pol_fraction = 100%, 100%\n",
    53	                encoding="utf-8",
    54	            )
    55	            counts["rl_written"] += 1
    56	
    57	    return counts

    62	    """Remove polarization includes from unpolarized generated process cards.
    63	
    64	    WHIZARD 3.1.8 rejects the zero-polarization block used in the initial scaffold.
    65	    Unpolarized cards should omit polarization.inc entirely.
    66	    """

   155	def localize_card(
   156	    *,
   157	    source_card: Path,
   158	    source_common: Path,
   159	    run_dir: Path,
   160	    events: int,
   161	    iterations: str,
   162	    sample_basename: str,
   163	) -> Path:
   164	    if not source_card.is_file():
   165	        raise BridgeError(f"missing source card: {source_card}")
   166	    if not source_common.is_dir():
   167	        raise BridgeError(f"missing source common directory: {source_common}")
   168	
   169	    run_dir.mkdir(parents=True, exist_ok=False)
   170	
   171	    shutil.copy2(source_card, run_dir / "process.original.sin")
   172	    shutil.copytree(source_common, run_dir / "common")
   173	
   174	    src = run_dir / "process.original.sin"
   175	    dst = run_dir / "process.bridge.sin"
   176	
   177	    text = src.read_text(encoding="utf-8")
   178	
   179	    def repl_include(match: re.Match[str]) -> str:
   180	        return f'include("common/{Path(match.group(1)).name}")'
   181	
   182	    text = re.sub(
   183	        r'(?m)^\s*include\("([^"]*/common/[^"]+)"\)\s*$',
   184	        repl_include,
   185	        text,
   186	    )
   187	
   188	    kept: list[str] = []
   189	    for line in text.splitlines():
   190	        m = re.match(r'\s*include\("common/([^"]+)"\)\s*$', line)
   191	        if m:
   192	            inc = run_dir / "common" / m.group(1)
   193	            if inc.exists() and inc.read_text(encoding="utf-8").strip() == "":
   194	                print(f"DROPPING_EMPTY_INCLUDE={inc}")
   195	                continue
   196	        kept.append(line)
   197	
   198	    dst.write_text("\n".join(kept) + "\n", encoding="utf-8")
   199	
   200	    integration = run_dir / "common" / "integration.inc"
   201	    if not integration.is_file():
   202	        raise BridgeError(f"missing integration include: {integration}")
   203	
   204	    t = integration.read_text(encoding="utf-8")
   205	    t, n_iter = re.subn(
   206	        r"(?m)^(\s*iterations\s*=\s*).*$",
   207	        rf"\g<1>{iterations}",
   208	        t,
   209	    )
   210	    if n_iter == 0:
   211	        t = f"iterations = {iterations}\n" + t
   212	    integration.write_text(t, encoding="utf-8")
   213	
   214	    event_output = run_dir / "common" / "event_output.inc"
   215	    if not event_output.is_file():
   216	        raise BridgeError(f"missing event-output include: {event_output}")
   217	
   218	    t = event_output.read_text(encoding="utf-8")
   219	    t, n_events = re.subn(
   220	        r"(?m)^(\s*n_events\s*=\s*).*$",
   221	        rf"\g<1>{events}",
   222	        t,
   223	    )
   224	    t, n_sample = re.subn(
   225	        r'(?m)^(\s*\$sample\s*=\s*).*$',
   226	        rf'\g<1>"{sample_basename}"',
   227	        t,
   228	    )
   229	
   230	    if n_events == 0:
   231	        raise BridgeError(f"could not find n_events assignment in {event_output}")
   232	    if n_sample == 0:
   233	        raise BridgeError(f"could not find $sample assignment in {event_output}")
   234	
   235	    event_output.write_text(t, encoding="utf-8")
   236	
   237	    return dst

   276	def write_sha256sums(run_dir: Path) -> None:
   277	    rows: list[tuple[str, str]] = []
   278	
   279	    for path in sorted(run_dir.rglob("*")):
   280	        if not path.is_file():
   281	            continue
   282	        rel = path.relative_to(run_dir).as_posix()
   283	        if rel == "SHA256SUMS.txt":
   284	            continue
   285	
   286	        h = hashlib.sha256()
   287	        with path.open("rb") as f:
   288	            for chunk in iter(lambda: f.read(1024 * 1024), b""):
   289	                h.update(chunk)
   290	        rows.append((h.hexdigest(), rel))
   291	
   292	    with (run_dir / "SHA256SUMS.txt").open("w", encoding="utf-8") as out:
   293	        for digest, rel in rows:
   294	            out.write(f"{digest}  {rel}\n")

    22	def derive_label(card: Path, generated_root: Path) -> str:
    23	    rel = card.relative_to(generated_root)
    24	    family = rel.parts[0]
    25	    channel_dir = rel.parts[1]
    26	
    27	    prefix = "f6f365_ee_"
    28	    family_tail = family[len(prefix):] if family.startswith(prefix) else family
    29	
    30	    if "_" in family_tail:
    31	        pol, family_channel = family_tail.split("_", 1)
    32	    else:
    33	        pol, family_channel = "unknown", family_tail
    34	
    35	    clean_channel = channel_dir.replace("__", "_")
    36	
    37	    if channel_dir == family_channel:
    38	        return family_tail
    39	
    40	    if channel_dir.startswith(family_channel + "__"):
    41	        return f"{pol}_{clean_channel}"
    42	
    43	    return f"{family_tail}_{clean_channel}"

    46	def derive_category(label: str) -> str:
    47	    pol = "polarized" if label.startswith(("LR100", "RL100")) else "unpolarized"
    48	    topology = "semileptonic" if "jets" in label else "dilepton"
    49	    return f"{topology}_{pol}"

    52	def discover_cards(generated_root: Path, mode: str) -> list[Path]:
    53	    if mode == "representative":
    54	        cards = [generated_root / rel for rel in REPRESENTATIVE_RELATIVE_CARDS]

    62	    if not cards:
    63	        raise FileNotFoundError(f"no process.sin cards found under {generated_root}")
    64	    return cards

    20	def load_manifest(path: Path) -> list[dict]:
    21	    records = []
    22	    with path.open("r", encoding="utf-8") as f:
    23	        for line_number, line in enumerate(f, start=1):
    24	            stripped = line.strip()
    25	            if not stripped or stripped.startswith("#"):
    26	                continue
    27	            try:
    28	                records.append(json.loads(stripped))
    29	            except json.JSONDecodeError as exc:
    30	                raise RuntimeError(f"{path}:{line_number}: invalid JSONL: {exc}") from exc
    31	    return records

    20	def main() -> int:

    23	    parser.add_argument("--output", type=Path, required=True)
    24	    parser.add_argument("--label", action="append", required=True)
    25	    parser.add_argument("--events-per-shard", type=int, default=500)

    57	            record["iterations"] = args.iterations if args.iterations is not None else base.get("iterations", "5:50000")
    58	            record["pythia_profile"] = args.pythia_profile
