   314	ensure_directory() {
   315	  local directory=$1
   316	  local attempt
   317	
   318	  # Normally all persistent campaign directories are prepared serially by
   319	  # submit_production.py.  This is a fallback for manually launched jobs and
   320	  # transient EOS namespace visibility.
   321	  if [[ -d "$directory" ]]; then
   322	    return 0
   323	  fi
   324	
   325	  for attempt in 1 2 3 4 5; do
   326	    # EOS/FUSE can report EEXIST during concurrent recursive creation even
   327	    # when the directory is being created successfully by another client.
   328	    mkdir -p -- "$directory" 2>/dev/null || true
   329	
   330	    if [[ -d "$directory" ]]; then
   331	      return 0
   332	    fi
   333	
   334	    sleep "$attempt"
   335	  done
   336	
   337	  echo "ERROR: unable to prepare persistent directory after retries: $directory" >&2
   338	  return 1
   339	}
