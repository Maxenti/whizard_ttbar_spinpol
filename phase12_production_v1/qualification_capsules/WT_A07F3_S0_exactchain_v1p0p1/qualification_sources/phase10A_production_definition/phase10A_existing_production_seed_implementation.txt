   235	def validate_configs(configs: Sequence[SampleConfig]) -> None:
   236	    if not configs:
   237	        raise ValueError("configuration contains no enabled samples")
   238	    campaigns = {cfg.campaign_id for cfg in configs}
   239	    if len(campaigns) != 1:
   240	        raise ValueError(f"configuration must contain exactly one campaign_id, got {campaigns}")
   241	    seen_ids: set[str] = set()
   242	    used_seeds: dict[int, str] = {}
   243	    for cfg in configs:
   244	        if cfg.sample_id in seen_ids:
   245	            raise ValueError(f"duplicate sample_id: {cfg.sample_id}")
   246	        seen_ids.add(cfg.sample_id)
   247	        if cfg.initial_state not in {"ee", "mumu"}:
   248	            raise ValueError(f"unsupported initial_state for {cfg.sample_id}: {cfg.initial_state}")
   249	        if cfg.decay_channel not in {"epmum", "mupem"}:
   250	            raise ValueError(f"unsupported decay_channel for {cfg.sample_id}: {cfg.decay_channel}")
   251	        if cfg.polarization not in {"LR100", "RL100"}:
   252	            raise ValueError(f"unsupported polarization for {cfg.sample_id}: {cfg.polarization}")
   253	        if cfg.beam1_helicity not in {-1, 1} or cfg.beam2_helicity not in {-1, 1}:
   254	            raise ValueError(f"beam helicities must be +/-1 for {cfg.sample_id}")
   255	        if not 0.0 <= cfg.beam1_pol_fraction <= 1.0:
   256	            raise ValueError(f"invalid beam1_pol_fraction for {cfg.sample_id}")
   257	        if not 0.0 <= cfg.beam2_pol_fraction <= 1.0:
   258	            raise ValueError(f"invalid beam2_pol_fraction for {cfg.sample_id}")
   259	        if cfg.n_shards <= 0 or cfg.events_per_shard <= 0:
   260	            raise ValueError(f"n_shards and events_per_shard must be positive for {cfg.sample_id}")
   261	        if cfg.request_cpus <= 0 or cfg.request_memory_mb <= 0:
   262	            raise ValueError(f"invalid Condor resources for {cfg.sample_id}")
   263	        if cfg.isr_mass_GeV <= 0 or cfg.isr_alpha <= 0:
   264	            raise ValueError(f"invalid ISR mass/alpha for {cfg.sample_id}")
   265	        if not cfg.isr_enabled:
   266	            raise ValueError(f"production-v1 requires ISR enabled for {cfg.sample_id}")
   267	        expected_spin_token = "_sc_" if cfg.spin_correlated else "_iso_"
   268	        if expected_spin_token not in cfg.sample_id:
   269	            raise ValueError(
   270	                f"sample_id spin token does not match spin_correlated={cfg.spin_correlated}: "
   271	                f"{cfg.sample_id}"
   272	            )
   273	        for shard in range(cfg.n_shards):
   274	            seed = cfg.shard_seed(shard)
   275	            owner = used_seeds.get(seed)
   276	            if owner is not None:
   277	                raise ValueError(f"duplicate seed {seed}: {owner} and {cfg.sample_id}/shard_{shard:04d}")
   278	            used_seeds[seed] = f"{cfg.sample_id}/shard_{shard:04d}"

   281	def filter_configs(configs: Sequence[SampleConfig], patterns: Sequence[str]) -> list[SampleConfig]:
   282	    if not patterns:
   283	        return list(configs)
   284	    import fnmatch
   285	
   286	    selected = [cfg for cfg in configs if any(fnmatch.fnmatch(cfg.sample_id, p) for p in patterns)]
   287	    if not selected:
   288	        raise ValueError(f"no samples matched patterns: {patterns}")
   289	    return selected
