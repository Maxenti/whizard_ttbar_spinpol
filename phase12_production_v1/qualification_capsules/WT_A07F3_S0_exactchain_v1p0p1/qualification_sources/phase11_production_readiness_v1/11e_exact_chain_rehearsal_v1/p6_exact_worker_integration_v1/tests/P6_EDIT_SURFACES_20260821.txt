   238	def validate_contiguous_daughter_ranges(
   239	    particles: list[Particle],
   240	    *,
   241	    event_number: int,
   242	) -> None:
   243	    """Require each parent's explicit daughter indices to form one range.
   244	
   245	    PYTHIA stores daughter relationships as inclusive first/last index ranges.
   246	    A non-contiguous child set therefore cannot be represented without
   247	    accidentally claiming unrelated particles as daughters.
   248	    """
   249	
   250	    children_by_parent: dict[int, list[int]] = {
   251	        index: [] for index in range(1, len(particles) + 1)
   252	    }
   253	
   254	    for child_index, particle in enumerate(particles, start=1):
   255	        parents = {
   256	            mother
   257	            for mother in (particle.mother1, particle.mother2)
   258	            if mother != 0
   259	        }
   260	        for parent in parents:
   261	            children_by_parent[parent].append(child_index)
   262	
   263	    for parent, children in children_by_parent.items():
   264	        if not children:
   265	            continue
   266	        ordered = sorted(set(children))
   267	        expected = list(range(ordered[0], ordered[-1] + 1))
   268	        if ordered != expected:
   269	            raise CanonicalizationError(
   270	                f"event {event_number}: parent {parent} has non-contiguous "
   271	                f"daughters {ordered}; implied range would be {expected}"
   272	            )

   274	def validate_canonical_particles(
   275	    particles: list[Particle],
   276	    *,
   277	    event_number: int,
   278	    closure_rel_tolerance: float,
   279	    closure_abs_tolerance_gev: float,
   280	) -> tuple[float, float]:
   281	    if len(particles) < 4:
   282	        raise CanonicalizationError(
   283	            f"event {event_number}: canonical event contains only {len(particles)} particles"
   284	        )
   285	
   286	    incoming = [p for p in particles if p.status == -1]
   287	    if len(incoming) != 2:
   288	        raise CanonicalizationError(
   289	            f"event {event_number}: expected exactly two ISTUP=-1 particles, found {len(incoming)}"
   290	        )
   291	    if any(p.status == -9 for p in particles):
   292	        raise CanonicalizationError(
   293	            f"event {event_number}: ISTUP=-9 remains after canonicalization"
   294	        )
   295	
   296	    size = len(particles)
   297	    for new_index, particle in enumerate(particles, start=1):
   298	        for label, mother in (("mother1", particle.mother1), ("mother2", particle.mother2)):
   299	            if mother < 0 or mother > size:
   300	                raise CanonicalizationError(
   301	                    f"event {event_number}: particle {new_index} {label}={mother} "
   302	                    f"outside [0,{size}]"
   303	                )
   304	            if mother == new_index:
   305	                raise CanonicalizationError(
   306	                    f"event {event_number}: particle {new_index} is its own mother"
   307	                )
   308	
   309	    validate_contiguous_daughter_ranges(
   310	        particles,
   311	        event_number=event_number,
   312	    )
   313	
   314	    max_abs, max_rel = closure_metrics(particles)
   315	    if max_abs > closure_abs_tolerance_gev and max_rel > closure_rel_tolerance:
   316	        raise CanonicalizationError(
   317	            f"event {event_number}: four-momentum closure failed: "
   318	            f"max_abs={max_abs:.6e} GeV max_rel={max_rel:.6e}"
   319	        )
   320	    return max_abs, max_rel

   323	def canonicalize_event(
   324	    event_lines: list[str],
   325	    *,
   326	    event_number: int,
   327	    closure_rel_tolerance: float,
   328	    closure_abs_tolerance_gev: float,
   329	) -> EventResult:
   330	    if not event_lines or event_lines[0].strip() != "<event>":

================================================================
CANONICALIZER TRANSFORM
================================================================
   330	    if not event_lines or event_lines[0].strip() != "<event>":
   331	        raise CanonicalizationError(f"event {event_number}: missing <event> opener")
   332	    if event_lines[-1].strip() != "</event>":
   333	        raise CanonicalizationError(f"event {event_number}: missing </event> closer")
   334	
   335	    cursor = 1
   336	    while cursor < len(event_lines) - 1 and not event_lines[cursor].strip():
   337	        cursor += 1
   338	    if cursor >= len(event_lines) - 1:
   339	        raise CanonicalizationError(f"event {event_number}: missing event header")
   340	
   341	    header_tokens = event_lines[cursor].split()
   342	    if len(header_tokens) < 6:
   343	        raise CanonicalizationError(
   344	            f"event {event_number}: malformed event header {event_lines[cursor]!r}"
   345	        )
   346	    try:
   347	        nup = int(header_tokens[0])
   348	    except ValueError as exc:
   349	        raise CanonicalizationError(
   350	            f"event {event_number}: invalid NUP value {header_tokens[0]!r}"
   351	        ) from exc
   352	
   353	    particle_start = cursor + 1
   354	    particle_end = particle_start + nup
   355	    if particle_end > len(event_lines) - 1:
   356	        raise CanonicalizationError(
   357	            f"event {event_number}: NUP={nup} exceeds available particle lines"
   358	        )
   359	
   360	    particles = [
   361	        parse_particle(event_lines[particle_start + offset], offset + 1)
   362	        for offset in range(nup)
   363	    ]
   364	    trailing = event_lines[particle_end:-1]
   365	
   366	    beams = [p for p in particles if p.status == -9]
   367	    bridges = [p for p in particles if p.status == -1]
   368	
   369	    if not beams and len(bridges) == 2:
   370	        rewired_isr_photons = normalize_isr_photon_mothers(particles)
   371	        max_abs, max_rel = validate_canonical_particles(
   372	            particles,
   373	            event_number=event_number,
   374	            closure_rel_tolerance=closure_rel_tolerance,
   375	            closure_abs_tolerance_gev=closure_abs_tolerance_gev,
   376	        )
   377	
   378	        if rewired_isr_photons == 0:
   379	            rendered = event_lines
   380	        else:
   381	            rendered = [
   382	                "<event>",
   383	                " " + " ".join(header_tokens),
   384	                *(particle.render() for particle in particles),
   385	                *trailing,
   386	                "</event>",
   387	            ]
   388	
   389	        return EventResult(
   390	            lines=rendered,
   391	            canonicalized=False,
   392	            removed_bridges=0,
   393	            promoted_beams=0,
   394	            rewired_isr_photons=rewired_isr_photons,
   395	            max_abs_closure_gev=max_abs,
   396	            max_rel_closure=max_rel,
   397	        )
   398	
   399	    if len(beams) != 2 or len(bridges) != 2:
   400	        raise CanonicalizationError(
   401	            f"event {event_number}: unsupported incoming layout: "
   402	            f"ISTUP=-9 count={len(beams)}, ISTUP=-1 count={len(bridges)}"
   403	        )
   404	
   405	    beam_indices = {p.original_index for p in beams}
   406	    bridge_to_beam: dict[int, int] = {}
   407	
   408	    for bridge in bridges:
   409	        if bridge.mother1 not in beam_indices or bridge.mother2 != 0:
   410	            raise CanonicalizationError(
   411	                f"event {event_number}: bridge particle {bridge.original_index} "
   412	                f"has mothers ({bridge.mother1},{bridge.mother2}), expected one beam mother"
   413	            )
   414	        beam = next(p for p in beams if p.original_index == bridge.mother1)
   415	        if beam.pid != bridge.pid:
   416	            raise CanonicalizationError(
   417	                f"event {event_number}: beam/bridge PID mismatch: "
   418	                f"beam {beam.original_index} pid={beam.pid}, "
   419	                f"bridge {bridge.original_index} pid={bridge.pid}"
   420	            )
   421	        bridge_to_beam[bridge.original_index] = beam.original_index
   422	
   423	    if len(set(bridge_to_beam.values())) != 2:
   424	        raise CanonicalizationError(
   425	            f"event {event_number}: bridge particles do not map one-to-one onto beams"
   426	        )
   427	
   428	    remove_indices = set(bridge_to_beam)
   429	    survivors = [p for p in particles if p.original_index not in remove_indices]
   430	    old_to_new = {
   431	        particle.original_index: new_index
   432	        for new_index, particle in enumerate(survivors, start=1)
   433	    }
   434	
   435	    for particle in survivors:
   436	        if particle.original_index in beam_indices:
   437	            particle.status = -1
   438	            particle.mother1 = 0
   439	            particle.mother2 = 0
   440	        else:
   441	            particle.mother1 = remap_mother(
   442	                particle.mother1,
   443	                old_to_new=old_to_new,
   444	                bridge_to_beam=bridge_to_beam,
   445	            )
   446	            particle.mother2 = remap_mother(
   447	                particle.mother2,
   448	                old_to_new=old_to_new,
   449	                bridge_to_beam=bridge_to_beam,
   450	            )
   451	
   452	    rewired_isr_photons = normalize_isr_photon_mothers(survivors)
   453	
   454	    max_abs, max_rel = validate_canonical_particles(
   455	        survivors,
   456	        event_number=event_number,
